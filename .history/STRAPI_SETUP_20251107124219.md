### Step 2: Set Up Strapi CMS (10 minutes)

#### Option A: Deploy Strapi on Railway (Recommended - Easiest)

1. **Go to Railway**: https://railway.app
2. **Sign up** with GitHub
3. **Create New Project** → **Deploy Strapi**
4. Railway will automatically:
   - Deploy Strapi
   - Set up PostgreSQL database
   - Provide you with a URL

5. **Save your Strapi URL**: Something like `https://your-app.railway.app`

6. **Access Strapi Admin**:
   - Go to `https://your-app.railway.app/admin`
   - Create your first admin account

#### Option B: Run Strapi Locally First (Alternative)

```bash
# In your project root
npx create-strapi-app@latest backend --quickstart

# This will:
# - Create a backend folder
# - Install Strapi
# - Start Strapi on http://localhost:1337
```

Create `.env.local` in your `frontend` directory:

```env

# OR for local: http://localhost:1337
STRAPI_API_TOKEN=your-api-token-here

# We'll get the API token from Strapi later
```

### Step 6: Configure Strapi Permissions

1. Go to **Settings** → **Users & Permissions Plugin** → **Roles** → **Public**
2. Enable these permissions for each collection:
   - ✅ find (list all)
   - ✅ findOne (get single item)
3. Save

This allows your Next.js site to read data from Strapi.

### Step 7: Create API Token

1. Go to **Settings** → **API Tokens**
2. Click **Create new API Token**
3. Name: `Next.js Frontend`
4. Token duration: `Unlimited`
5. Token type: `Read-only`
6. Copy the token and add it to your `.env.local`

### Step 8: Test the Connection

Create `frontend/app/api/test/route.ts`:

```typescript
import { NextResponse } from 'next/server'

export async function GET() {
  const strapiUrl = process.env.NEXT_PUBLIC_STRAPI_URL
  
  try {
    const response = await fetch(`${strapiUrl}/api/internships`)
    const data = await response.json()
    
    return NextResponse.json({
      success: true,
      message: 'Connected to Strapi!',
      internships: data.data.length
    })
  } catch (error) {
    return NextResponse.json({
      success: false,
      error: 'Could not connect to Strapi'
    }, { status: 500 })
  }
}
```

Test it: Run `npm run dev` and visit `http://localhost:3000/api/test`

### Step 9: Create Strapi Helper Functions

Create `frontend/lib/strapi.ts`:

```typescript
interface StrapiResponse<T> {
  data: T
  meta?: any
}

const STRAPI_URL = process.env.NEXT_PUBLIC_STRAPI_URL || 'http://localhost:1337'

export async function fetchFromStrapi<T>(
  endpoint: string,
  options?: RequestInit
): Promise<T> {
  const url = `${STRAPI_URL}/api${endpoint}`
  
  const response = await fetch(url, {
    ...options,
    headers: {
      'Content-Type': 'application/json',
      ...options?.headers,
    },
  })

  if (!response.ok) {
    throw new Error(`Strapi fetch error: ${response.statusText}`)
  }

  const data: StrapiResponse<T> = await response.json()
  return data.data
}

// Helper functions for each content type
export async function getInternships() {
  return fetchFromStrapi('/internships?populate=*')
}

export async function getInternshipBySlug(slug: string) {
  return fetchFromStrapi(`/internships?filters[slug][$eq]=${slug}&populate=*`)
}

export async function getProjects() {
  return fetchFromStrapi('/projects?populate=*&sort=order:asc')
}

export async function getBlogPosts() {
  return fetchFromStrapi('/blog-posts?populate=*&sort=publishedDate:desc')
}

export async function getBlogPostBySlug(slug: string) {
  return fetchFromStrapi(`/blog-posts?filters[slug][$eq]=${slug}&populate=*`)
}
```

## Troubleshooting

**Can't connect to Strapi?**
- Check your `NEXT_PUBLIC_STRAPI_URL` in `.env.local`
- Make sure Strapi is running
- Check Strapi permissions are set to Public

**CORS errors?**
- In Strapi: `config/middlewares.js` add your Next.js URL to allowed origins
